typedef struct
{
  short* drawlist;
  int h;
  float projection;
  FVector* MeshPoints;
  int MeshW;
  int MeshH;
  Fx8* BitMap;
  int cam_x;
  int cam_y;
} DoEffectArguments;

static int DoEffect(DoEffectArguments* args)
{
  short* drawlist = args->drawlist;
  int h = args->h;
  float projection = args->projection;
  FVector* MeshPoints = args->MeshPoints;
  int MeshW = args->MeshW;
  int MeshH = args->MeshH;
  Fx8* BitMap = args->BitMap;
  int PosX = args->cam_x;
  int PosY = args->cam_y;
  
  short* ori = drawlist;
  
  int half_h = h >> 1;
  
  for (int iy = 0; iy < MeshH; iy++) 
  { 
    int MapPosY = (PosY + iy) & 127; 
    FVector* LinePoints = &MeshPoints[iy * MeshW]; 
    Fx8* BitMapLine = &BitMap[MapPosY * 128]; 
    float z = LinePoints->z;
    Fx8 proj = (int)(projection * z * 256.0f);
    
    for (int ix = 0; ix < MeshW; ix++) 
    { 
      FVector* Point = LinePoints; 
      Fx8 y = Point->y;
      int x = Point->x;
      int MapPosX = (PosX + ix) & 127;
  
      y -= BitMapLine[MapPosX];
  
      int yp = half_h + ((y * proj) >> 16);
  
      drawlist[0] = x;
      drawlist[1] = yp;
      drawlist[2] = 0xff;
      drawlist += 3;
      
      LinePoints++;
    }
  }
  
  return (drawlist - ori);
}

typedef struct
{
  unsigned int* pixels;
  int stride;
  unsigned int w;
  unsigned int h;
} screen_desc_t;

typedef struct
{
  unsigned int* pixels;
  unsigned int w;
  unsigned int h;
} map_desc_t;

static inline void plot(screen_desc_t* screen, unsigned int x, unsigned int y, unsigned int color)
{
  if (x < 0 || x >= screen->h) return;
  if (y < 0 || y >= screen->w) return;
  
  screen->pixels[x + y * screen->stride] = color;
}

#define MAX_DEPTH 1400

static int Voxels(screen_desc_t* screen, camera_desc_t* camera, map_desc_t* map)
{
  int x_screen;
  int z_map;
  int n_draw = 0;
  float y;
  float y_center = (float)(screen->h >> 1);

  float angle = 90.0f - (FOV * 0.5f);
  float radians_angle = angle * 3.141592f/180.0f;
  float sin_angle = sinf(radians_angle);
  float cos_angle = cosf(radians_angle);
  
  // calculate x depending on the FOV
  float max_vector_x = cos_angle * (1.0f / sin_angle);
  float min_vector_x = -max_vector_x;

  int mip_level = 0;

  for (x_screen = 0; x_screen < screen->w; x_screen++)
  {
    // lerp -1 to 1
    int v_map = camera->pos_z;
    float x_map = (float)camera->pos_x;
    float step_x = lerpf(min_vector_x, max_vector_x, (float)x_screen / (float)screen->w);

    int last_y = screen->h;
    int max_y = screen->h;

    for (z_map = 1; z_map < MAX_DEPTH; z_map++)
    {
      mip_level = z_map / 1000;

      int v = ((int)x_map) & (map[0].w - 1);
      int u = v_map & (map[0].h - 1);

      v = v >> mip_level;
      u = u >> mip_level;

      unsigned int pixel = map[mip_level].pixels[u + v * map[mip_level].w];

      unsigned int h = pixel >> 24;
      unsigned int color = (pixel & 0x00ffffff) | 0xff000000;

      y = (float)(h - camera->h);
      
      x_map += step_x;
      v_map += 1;

      int proj_x = x_screen;
      int proj_y = y_center - (y * camera->proj / (float)z_map);

      if (proj_y < max_y)
      {
        for (; max_y > proj_y; max_y--)
        {
          /** Inverted plot */
          plot(screen, max_y, proj_x, color);
          
          /** Non-Inverted plot */
          //plot(screen, proj_x, max_y, color);
        }
      }
    }
  }
  return MAX_DEPTH * screen->w;
}

static int Rotate(unsigned int* src, unsigned int* dst, int src_w, int src_h, int src_stride, int dst_w, int dst_h, int dst_stride)
{
  for (int y = 0; y < src_h; y++)
  {
    for (int x = 0; x < src_w; x++)
    {
      unsigned int tmp = src[x + y * src_stride];
      src[x + y * src_stride] = dst[y + x * dst_stride];
      dst[y + x * dst_stride] = tmp;
    }
  }
  return 0;
}

static int TilingRotate(unsigned int* src, unsigned int* dst, int src_w, int src_h, int src_stride, int dst_w, int dst_h, int dst_stride)
{
  assert((src_w % TLS) == 0);
  assert((src_h % TLS) == 0);

  for(int y = 0; y < src_h; y += TLS)
  {
    for(int x = 0; x < src_w; x += TLS)
    {
      Rotate(src + (x + y * src_w), dst + (y + x * dst_w), TLS, TLS, src_stride, TLS, TLS, dst_stride);
    }
  }

  return src_w * src_h;
}
